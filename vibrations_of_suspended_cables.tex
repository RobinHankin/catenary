\begin{align}\tag{2.2a}\label{2.2a}
  \da{x}{s}  &= \da{s}{s}\db{x}{s} = \da{s}{s}\cos\overline{\alpha}\cos\overline{\gamma}\\
\tag{2.2b}\label{2.2b}
  \da{y}{s}  &= \da{s}{s}\db{y}{s} = \da{s}{s}\sin\overline{\alpha}\\
\tag{2.2c}\label{2.2c}
  \da{z}{x}  &= \da{s}{s}\db{z}{s} = \da{s}{s}\cos\overline{\alpha}\sin\overline{\gamma}
\end{align}

equating 2.1 and 2.2 gives:

\begin{align}
  \da{s}{s}\cos\overline{\alpha}\cos\overline{\gamma} &=
  (1 + u_s -v\alpha_s)\cos\alpha
  -(v_s     +u\alpha_s)\sin\alpha\\
\tag{2.3b}\label{2.3b}
\da{s}{s}\sin\overline{\alpha}  &= 
  (1 + u_s -v\alpha_s)\sin\alpha
+(v_s     +u\alpha_s)\cos\alpha\\
\tag{2.3c}\label{2.3c}
\da{s}{s}\cos\overline{\alpha}\sin\overline{\gamma} &=
w_s
\end{align}

Eliminating $\overline{\alpha},\overline{\gamma}$ gives

\begin{equation}\tag{2.4}\label{2.4}
  \da{s}{s}=\sqrt{
    w_s^2 + (1 + u_s -v\alpha_s)^2+(v_s+u\alpha_s)^2}
\end{equation}

Define the longitudinal stress by

$$\overline{\epsilon}=\frac{d\overline{s}-ds}{ds}$$

Thus

\begin{equation}\label{2.5}\tag{2.5}
  \overline{\epsilon} = \sqrt{w_s^2 + (1 + u_s -v\alpha_s)^2+(v_s+u\alpha_s)^2}-1
\end{equation}

(and because $\overline{\epsilon}=0$ for the inextensible case we have 

$$w_s^2 + (1 + u_s -v\alpha_s)^2+(v_s+u\alpha_s)^2=1$$


Now potential energy:

\begin{equation}\label{2.6}\tag{2.6}
  \frac{\partial V_g}{\partial s}  = \frac{\partial V_{0g}}{\partial s} +
    \rho g(u\sin\alpha + v\cos\alpha)
\end{equation}

And kinetic energy:

\begin{equation}\label{2.7}\tag{2.7}
  \frac{\partial\mathrm{KE}}{\partial s}=\frac{\rho}{2}\left(
  u_t^2 +  v_t^2 +  w_t^2\right)
  \end{equation}



\begin{align}
  \overline{u}_t &= \tag{2.8a}\label{2.8a}
  \left(u_t\cos\beta + v_t\sin\beta\right)\cdot
  \left(\cos\overline{\gamma}+(1-\cos\overline{\gamma})\sin^2\overline{\alpha}\right)\nonumber\\
  &{\qquad} +
  \left(-u_t\sin\beta + v_t\cos\beta\right)\cdot
  \left(1-\cos\overline{\beta}\right)\sin\overline{\alpha}\cos\overline{\alpha}\nonumber\\
  &{\qquad} +
  w_t\cdot\sin\overline{\gamma}\cdot\cos\overline{\alpha}\nonumber\\
  \overline{v}_t &= \tag{2.8b}\label{2.8b}
  \left(u_t\cos\beta + v_t\sin\beta\right)\cdot\left(1-\cos\overline{\gamma}\right)\cdot
  \sin\overline{\alpha}\cos\overline{\alpha}\nonumber\\
  &{\qquad} +
  (u_t\sin\beta + v_t\cos\beta)\cdot\left(\cos\overline{\gamma} + (1-\cos\overline{\gamma})
  \cos^2\overline{\alpha}\right) - w_t\sin\overline{\gamma}\sin\overline{\alpha}\nonumber\\
  \overline{w}_t &= \tag{2.8c}\label{2.8c}
  -\left(u_t\cos\beta + v_t\sin\beta\right)\cdot
  (\sin\overline{\gamma}\cos\overline{\alpha})\nonumber\\
  &{\qquad} +
  (-u_t\sin\beta + v_t\cos\beta)\cdot\left(\sin\overline{\gamma}\sin\overline{\alpha}\right) +
  w_t\cos\overline{\gamma}\nonumber
\end{align}

Increment of virtual work $\delta W_f$:

\begin{align}\tag{2.9}\label{2.9}
  \delta W_f &= -\int_{s_1}^{s^2}
  \left(
  f_1\overline{u_t}\delta\overline{u} +
  f_2\overline{v_t}\delta\overline{v} +
  f_2\overline{w_t}\delta\overline{w}
  \right)\,ds\\
  &=   \int_{s_1}^{s^2}
  \left(\tag{2.9b}\label{2.9b}
  Q_u\delta u +Q_v\delta v+Q_w\delta w\right)\,ds
  \end{align}



Now the tension at any position is denoted $\so$.  The increment of
work of the internal forces on the cable as it is strained between two
states is

\begin{equation}\tag{2.10}\label{2.10}
  \delta W_s= -\int_{s_1}^{s_2}\so\delta\overline{\epsilon}\,ds
\end{equation}

where $\delta\overline{\epsilon}$ is an incremental change in the
strain.  Using equation~\ref{2.5}


\begin{equation}\tag{2.11}\label{2.11}
  \delta\overline{\epsilon} =
  \frac{
    (1+u_s-\alpha_s v)\cdot\left(\delta u_s-\alpha_s\delta v\right)
    + (v_s+\alpha_s u)\cdot\left(\delta v_s+\alpha_s\delta u\right)
    + w_s\delta w_s
  }{
    \sqrt{w_S^2 + (1+u_s-v\alpha_s)^2 + (v_s + u\alpha_s)^2}
  }
\end{equation}

Then substitute \ref{2.11} into \ref{2.10} [NB using \ref{2.5}]:

\begin{align}
  \delta W_s &=
  \ints\left[\embig

    &{}\qquad

    + \left[\so\frac{v_s+\alpha_su}{\eo+1}\right]_s\right\rbrace\,\delta v\nonumber\\
    &{}\qquad
    + \left\lbrace\left[\frac{\so w_s}{\eo+1}\right]_s\right\rbrace
    \left.\embig
    \right]\,ds\nonumber\\
  &{}\qquad
  -\left[
  \frac{\so}{\eo+1}

  (1+u_s-\alpha_sv)\delta u +(v_s+\alpha_su)\delta v+w_s\delta w

\end{align}
